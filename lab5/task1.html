function getProductDetails(productId, successCallback, errorCallback) {
  // Приклад бази товарів
  const products = [
    { id: 1, name: "Ноутбук", price: 25000 },
    { id: 2, name: "Смартфон", price: 15000 },
    { id: 3, name: "Навушники", price: 2000 }
  ];

  // Пошук товару
  const product = products.find(item => item.id === productId);

  if (product) {
    // Якщо товар знайдено — викликаємо successCallback
    successCallback(product);
  } else {
    // Якщо не знайдено — викликаємо errorCallback
    errorCallback(`Товар з id ${productId} не знайдено`);
  }
}

// Приклад використання
getProductDetails(
  2,
  (data) => console.log("Знайдено товар:", data),
  (error) => console.error("Помилка:", error)
);
