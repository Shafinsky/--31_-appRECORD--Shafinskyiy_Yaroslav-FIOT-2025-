2-snackbar.js

document.querySelector('.form').addEventListener('submit', e => {
  e.preventDefault();

  const delay = Number(e.target.delay.value);
  const state = e.target.state.value;

  const promise = new Promise((resolve, reject) => {
    setTimeout(() => {
      if (state === 'fulfilled') {
        resolve(delay);
      } else {
        reject(delay);
      }
    }, delay);
  });

  promise
    .then(ms => {
      iziToast.success({
        title: 'Success',
        message: `Fulfilled promise in ${ms}ms`,
        timeout: 2000,
        position: 'topRight',
      });
    })
    .catch(ms => {
      iziToast.error({
        title: 'Error',
        message: `Rejected promise in ${ms}ms`,
        timeout: 2000,
        position: 'topRight',
      });
    });

  e.target.reset();
});
