2-form.js

const form = document.querySelector('.feedback-form');
const STORAGE_KEY = 'feedback-form-state';

// 1️⃣ Глобальний об'єкт з пустими значеннями
let formData = {
  email: '',
  message: '',
};

// 2️⃣ Відновлення даних при завантаженні
const saved = localStorage.getItem(STORAGE_KEY);

if (saved) {
  const parsed = JSON.parse(saved);

  // підставляємо у форму, якщо є
  if (parsed.email) {
    form.email.value = parsed.email;
    formData.email = parsed.email;
  }

  if (parsed.message) {
    form.message.value = parsed.message;
    formData.message = parsed.message;
  }
}

// 3️⃣ Відстеження input (делегування)
form.addEventListener('input', e => {
  const field = e.target.name;
  const value = e.target.value.trim();

  formData[field] = value;

  localStorage.setItem(STORAGE_KEY, JSON.stringify(formData));
});

// 4️⃣ Submit
form.addEventListener('submit', e => {
  e.preventDefault();

  if (!formData.email || !formData.message) {
    alert('Fill please all fields');
    return;
  }

  console.log('Submitted data:', formData);

  // очищення
  form.reset();
  formData = { email: '', message: '' };
  localStorage.removeItem(STORAGE_KEY);
});
